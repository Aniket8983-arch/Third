import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

df=pd.read_csv('Mall_Customers.csv')
df.describe()

df.head()

df.shape

print(df.duplicated().sum())

df.isna().sum()

from sklearn.preprocessing import LabelEncoder
le=LabelEncoder()
df['Genre']=le.fit_transform(df['Genre'])

df.head()

data=df[['Annual Income (k$)','Spending Score (1-100)']]

from sklearn.cluster import KMeans

model1=KMeans(n_clusters=3,random_state=42,n_init=10)
model1.fit(data)
cluster=model1.predict(data)

model1.inertia_

model1.cluster_centers_

sns.scatterplot(x=data['Annual Income (k$)'],y=data['Spending Score (1-100)'],hue=cluster,palette='viridis',s=100)

centroids=model1.cluster_centers_

plt.scatter(centroids[:,0],centroids[:,1],marker='x',color='red',label='Centroids')
plt.legend(loc='upper right')
plt.show()

wcss=[]

for i in range(1,11):
    model2=KMeans(i,random_state=42,n_init=10)
    model2.fit(data)
    wcss.append(model2.inertia_)
plt.plot(range(1,11),wcss,marker='o',label='WCSS')
plt.title('Elbow Method')
plt.xlabel('No. of Clusters')
plt.ylabel('WCSS')
plt.xticks(range(1,11))
plt.show()

from scipy.cluster.hierarchy import linkage,dendrogram

z=linkage(data,method='ward')
plt.figure(figsize=(8,4))
dendrogram(z)

plt.title('Dendrogram')
plt.xlabel('groups')
plt.ylabel('distance')
