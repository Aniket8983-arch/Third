import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

df = pd.read_csv("Mall_Customers - Mall_Customers.csv")
df.head()
df.describe()
df.isna().sum()

plt.hist(df["Genre"])
plt

from sklearn.preprocessing import LabelEncoder
le = LabelEncoder()
df["Genre"] = le.fit_transform(df["Genre"])
df.head()

data = df[["Annual Income (k$)", "Spending Score (1-100)"]]
df.tail()

from sklearn.cluster import KMeans
import pandas as pd
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
kmeans.fit(data)
clusters = kmeans.predict(data)
print(kmeans.labels_)
kmeans.inertia_
kmeans.cluster_centers_
kmeans.n_clusters

sns.scatterplot(
    x=data["Annual Income (k$)"],
    y=data["Spending Score (1-100)"],
    hue=clusters, palette="viridis", s=100
)
centroids = kmeans.cluster_centers_
plt.scatter(centroids[:, 0], centroids[:, 1], marker="X", color="red", label="Centroids")
plt.legend(loc="upper right")
plt.show()

wcss = []
for i in range(1, 11):
    model = KMeans(i, random_state=42, n_init=10)
    model.fit(data)
    wcss.append(model.inertia_)

plt.plot(range(1, 11), wcss, marker="o", label="WCSS")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.title("Elbow Method")
plt.xticks(range(1, 11))
plt.grid(alpha=0.3)
plt.legend(loc="upper right")
plt.show()

kmeans = KMeans(n_clusters=5, random_state=42)
kmeans.fit(data)
clusters = kmeans.predict(data)
print(kmeans.labels_)
kmeans.cluster_centers_
kmeans.inertia_

plt.figure(figsize=(8, 4))
sns.scatterplot(
    x=data["Annual Income (k$)"],
    y=data["Spending Score (1-100)"],
    hue=clusters, palette="viridis", s=100
)
plt.scatter(
    kmeans.cluster_centers_[:, 0], kmeans.cluster_centers_[:, 1],
    s=100, marker="P", c='red', label="Centroids"
)
plt.legend(loc="upper right")

from scipy.cluster.hierarchy import linkage, dendrogram
Z = linkage(data, method="ward")
plt.figure(figsize=(8, 4))
dendrogram(Z)
plt.title("Dendrogram - Hierarchical Clustering")
plt.xlabel("Customer groups")
plt.ylabel("Distance")
plt.grid()
plt.show()
